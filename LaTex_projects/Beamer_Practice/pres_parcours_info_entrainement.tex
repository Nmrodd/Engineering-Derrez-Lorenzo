\section{Parcours en largeur}

\begin{frame}{\textbf{Parcours en largeur}}
    \begin{block}{Introduction}
        \begin{algorithm}[H]
            \footnotesize
            \caption{Parcours en largeur}
            \begin{algorithmic}[1]
                \Require racine
                \If{racine est vide}
                    \State \Return $[\,]$
                \Else
                    \State $F \gets \texttt{creer\_file()}$
                    \State \Call{enfiler}{racine, $F$}
                    \While{$F$ n'est pas vide}
                        \State $noeud \gets \Call{defiler}{F}$
                        \State \Call{traiter}{noeud}
                        \If{$noeud.gauche \neq nul$}
                            \State \Call{enfiler}{noeud.gauche, $F$}
                        \EndIf
                        \If{$noeud.droite \neq nul$}
                            \State \Call{enfiler}{noeud.droite, $F$}
                        \EndIf
                    \EndWhile
                \EndIf
            \end{algorithmic}
        \end{algorithm}
    \end{block}
\end{frame}

\subsection{Infixe}
\begin{frame}{\textbf{Parcours en profondeur Infixe}}
    \begin{block}{Introduction}
        \begin{algorithm}[H]
            \footnotesize
            \caption{DFS Infixe}
            \begin{algorithmic}[1]
                \Require racine
                \If{racine est vide}
                    \State \Return $[\,]$
                \Else
                    \State $resultats \gets$ liste vide
                    \State \Call{parcourirInfixe}{racine, resultats}
                    \State \Return resultats
                \EndIf
            \end{algorithmic}
        \end{algorithm}
    \end{block}
\end{frame}
